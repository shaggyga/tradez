def sha(raw):return hashlib.sha256(raw).hexdigest()
def epoch(value):
    try:
        if isinstance(value,(int,float)) and not isinstance(value,bool):return float(value)
        return datetime.fromisoformat(str(value).replace('Z','+00:00')).timestamp()
    except (ValueError,TypeError):return None
def age(value,now):
    stamp=epoch(value);return now-stamp if stamp is not None else None
def compact(value,keys):return {k:value.get(k) for k in keys if k in value}
def bounded_json(path,limit=2*1024*1024):
    with path.open('rb') as f:
        before=os.fstat(f.fileno());raw=f.read(limit+1);after=os.fstat(f.fileno())
    observed=time.time()
    if len(raw)>limit:raise ValueError('read_size_limit')
    if (before.st_ino,before.st_size,before.st_mtime_ns)!=(after.st_ino,after.st_size,after.st_mtime_ns):raise ValueError('changed_during_read')
    value=json.loads(raw)
    return value,{'path':str(path),'bytes':len(raw),'sha256':sha(raw),'observed_epoch':observed}
def source_bindings(value):
    bindings=value.get('source_bindings') or {};result=[]
    for name,expected in bindings.items():
        if not re.fullmatch(r'[A-Za-z0-9_]+\.py',name):result.append({'source':name,'error':'unexpected_source_path'});continue
        path=PROJECT/name
        try:actual=sha(path.read_bytes());result.append({'source':name,'expected_sha256':expected,'actual_sha256':actual,'matches':actual==expected})
        except OSError as e:result.append({'source':name,'error':type(e).__name__})
    return {'count':len(bindings),'all_match':bool(result) and all(r.get('matches') for r in result),'files':result}
def coverage(source):

    for row in rows[:68]:
        families={}
        for name,slot in (row.get('families') or {}).items():

            families[name]={**compact(slot,['status','reason','reason_label','observed_epoch']),
                'readiness':compact(ready,['status','reason','reason_label','observed_epoch','price_bar_close_epoch','news_evidence_epoch','news_expires_epoch','diagnostics']),
                'last_attempt':compact(attempt,['epoch','reason','attempt_id','bucket']),'counts':counts}
        arms=[compact(a,['family','issued_epoch','reference_epoch','target_epoch','publication_epoch','decision_id','forecast_sha256']) for a in row.get('active_forecasts') or []]
        row_results.append({**compact(row,['instrument','status','reason','reason_label','observed_epoch']),'families':families,'active_forecasts':arms})
    return {**compact(source,['study_version','status','reason','generated_epoch','observed_epoch','registry_sha256','summary_sha256','counts','family_counts','ledger_counts','worker_observation','publication_read_failures','retained_generation','research_only','can_place_orders','can_promote','can_authorize','account_eligible','proof_eligible','historical_rows_imported','primary_selection']),

        'actual_unelapsed_forecast_pairs':sum(any(a.get('target_epoch',0)>now for a in r.get('active_forecasts') or []) for r in rows),
