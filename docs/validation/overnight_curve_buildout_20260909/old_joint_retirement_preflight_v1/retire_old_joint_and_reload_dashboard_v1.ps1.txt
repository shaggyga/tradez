function Hash-File([string]$Path){(Get-FileHash -LiteralPath $Path -Algorithm SHA256).Hash.ToLowerInvariant()}
if((Hash-File $specification) -ne $SpecificationSha256){throw 'Specification hash mismatch'}
$spec=Get-Content -LiteralPath $specification -Raw | ConvertFrom-Json
function Verify-Sources {
    if((Hash-File $specification) -ne $SpecificationSha256){throw 'Specification changed'}
    foreach($binding in $spec.source_bindings.PSObject.Properties){
        $path=[IO.Path]::GetFullPath($binding.Name)
        if(-not $path.StartsWith($workspace+'\',[StringComparison]::OrdinalIgnoreCase)){throw 'Source path outside workspace'}
        if((Hash-File $path) -ne $binding.Value){throw 'Reviewed source or evidence changed'}
    }
    foreach($binding in $spec.registries.PSObject.Properties){
        $path=Join-Path (Join-Path $project 'config') $binding.Name
        if((Hash-File $path) -ne $binding.Value){throw 'Registry changed'}
        $registry=Get-Content -LiteralPath $path -Raw | ConvertFrom-Json
        if($registry.research_only -ne $true -or $registry.can_place_orders -ne $false -or $registry.can_promote -ne $false -or $registry.can_authorize -ne $false){throw 'Study authority mismatch'}
        foreach($source in $registry.source_bindings.PSObject.Properties){
            if($source.Name -notmatch '^(?:[a-z0-9_]+/)*[a-z0-9_]+\.py$'){throw 'Unexpected registered path'}
            if((Hash-File (Join-Path $project $source.Name)) -ne $source.Value){throw 'Frozen study source changed'}
        }
    }
}
function Project-Processes {
    @(Get-CimInstance Win32_Process | Where-Object {$_.Name -match '^(python|pythonw|pwsh|powershell)(\.exe)?$'} | ForEach-Object {
        $match=[regex]::Match([string]$_.CommandLine,'(?i)(?:"([^"]+\.(?:py|ps1))"|([^\s"]+\.(?:py|ps1)))(?:\s|$)')
        $path=if($match.Groups[1].Success){$match.Groups[1].Value}else{$match.Groups[2].Value}
        if($path.StartsWith($project+'\',[StringComparison]::OrdinalIgnoreCase)){
            [pscustomobject]@{Raw=$_;ScriptPath=$path}
        }
    })
}
function Identity($row){[ordered]@{pid=[int]$row.Raw.ProcessId;parent_pid=[int]$row.Raw.ParentProcessId;name=$row.Raw.Name;created_utc=$row.Raw.CreationDate.ToUniversalTime().ToString('o');script_path=$row.ScriptPath}}
function Same-Process($row){
    $now=@(Get-CimInstance Win32_Process -Filter ('ProcessId = '+[int]$row.Raw.ProcessId))
    if($now.Count -ne 1 -or $now[0].CreationDate -ne $row.Raw.CreationDate -or $now[0].Name -ne $row.Raw.Name -or $now[0].CommandLine -ne $row.Raw.CommandLine){throw 'Process identity changed'}
}
function Save-Receipt([string]$name,$value){

    if(Test-Path -LiteralPath $path){throw 'Receipt already exists'}
    $value | ConvertTo-Json -Depth 8 | Set-Content -LiteralPath $path -Encoding utf8
}
