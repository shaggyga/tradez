def sha(raw):return hashlib.sha256(raw).hexdigest()


def record(path):

def require(ok):

def permitted_evidence_path(path):
    """Compact receipts/docs/tests only; never private transport/data payloads."""
    path=Path(path)
    forbidden=lambda part:part.lower() in {'private','raw','raw_responses','db','databases','datasets'} or part.lower().startswith(('private_','raw_'))
    return not any(forbidden(part) for part in path.parts) and path.suffix.lower() in {'.json','.md','.xml','.py','.ps1','.patch','.txt'}
