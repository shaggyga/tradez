"""AST-only handler wiring review; no dashboard imports or server actions."""
import ast
import hashlib
from pathlib import Path
import re
import threading
from types import SimpleNamespace
from urllib.parse import urlparse

import pytest

ROOT = Path(__file__).resolve().parent
BASE = ROOT.parent / 'overnight_curve_buildout_20260909/coherent_reader_design/ledger_observer_v1'
DRAFT = BASE / 'dashboard_handler_draft.py.txt'


@pytest.mark.parametrize('path,compact', [('/api/main', True), ('/api/state', True),
                                          ('/api/joint-v3-ledger-observation', False)])
def test_namespace_is_separate_and_cached_main_state_never_mutated(path, compact):
    tree = ast.parse(DRAFT.read_text(encoding='utf-8'))
    node = next(function for cls in tree.body if isinstance(cls, ast.ClassDef)
                for function in cls.body if isinstance(function, ast.FunctionDef) and function.name == 'do_GET')
    calls, sent = [], []
    original = {'joint_price_news_forecasts': {'status': 'unavailable', 'reason': 'pair_summary_generation_mismatch'},
                'joint_collection_status': {'status': 'unavailable'}}
    def observed(*, compact=False):
        calls.append(compact)
        return {'schema_version': 'independent-test-observer', 'consumer': {'current_forecast_pairs': 66}}
    scope = {'urlparse': urlparse, 'current_joint_ledger_observation': observed}
    exec(compile(ast.Module(body=[node], type_ignores=[]), str(DRAFT), 'exec'), scope)
    request = SimpleNamespace(path=path, current_main_state=lambda: original, send_json=sent.append)
    scope['do_GET'](request)
    assert calls == [compact] and len(sent) == 1
    assert 'joint_v3_ledger_observation' not in original
    if compact:
        assert sent[0]['joint_v3_ledger_observation']['consumer']['current_forecast_pairs'] == 66
        assert sent[0]['joint_collection_status']['status'] == 'unavailable'
        assert sent[0]['joint_price_news_forecasts']['reason'] == 'pair_summary_generation_mismatch'
    else:
        assert sent[0]['schema_version'] == 'independent-test-observer'


def test_unset_deployment_pin_withholds_without_import_or_fabricated_status():
    tree = ast.parse(DRAFT.read_text(encoding='utf-8'))
    node = next(n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == 'current_joint_ledger_observation')
    scope = {'re': re, 'JOINT_LEDGER_OBSERVER_SPEC_SHA256': 'REPLACE_WITH_REVIEWED_SPEC_FILE_SHA256',
             '_JOINT_LEDGER_OBSERVER_API': None}
    exec(compile(ast.Module(body=[node], type_ignores=[]), str(DRAFT), 'exec'), scope)
    value = scope['current_joint_ledger_observation']()
    assert value['status'] == 'unavailable' and value['reason'] == 'observer_api_not_configured'
    assert value['consumer'] is None and value['can_place_orders'] is False


def test_draft_preserves_all_old_producer_and_status_functions_and_no_html_changes():
    original = (ROOT / 'oanda_practice_live_dashboard.py').read_bytes()
    assert hashlib.sha256(original).hexdigest() == 'cc0f6698a30c59cb291dada08d73283f0484996e6873b8ce27894e0168614f6f'
    before, after = ast.parse(original), ast.parse(DRAFT.read_text(encoding='utf-8'))
    names = ('summarize_joint_price_news_forecasts', '_summarize_registered_family_forecasts',
             'project_joint_collection_status', 'build_main_state')
    for name in names:
        get = lambda tree: next(n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == name)
        assert ast.dump(get(before)) == ast.dump(get(after))
