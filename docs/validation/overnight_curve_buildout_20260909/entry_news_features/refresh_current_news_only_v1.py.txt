def sha(raw):return hashlib.sha256(raw).hexdigest()


def require(value, reason):
    if not value:raise ValueError(reason)


def deny_io(event, args):
    if event.startswith('sqlite3.connect') or event in ('socket.connect','socket.getaddrinfo','subprocess.Popen','os.system'):
        raise RuntimeError('current_only_audit_forbids_database_network_or_process')


def load_json(raw):
    def pairs(values):
        result={}
        for key,value in values:
            require(key not in result,'duplicate_json_key');result[key]=value
        return result
    return json.loads(raw,object_pairs_hook=pairs,
        parse_constant=lambda _:(_ for _ in ()).throw(ValueError('nonfinite_json')))


def identity(info):
    return dict(device=info.st_dev,inode=info.st_ino,bytes=info.st_size,mtime_ns=info.st_mtime_ns)


def bounded_read(path, maximum=MAX_BYTES, *, clock=time.time):
    started=clock();before=path.stat()
    require(not getattr(before,'st_file_attributes',0)&0x400,'reparse_source_refused')
    require(0<before.st_size<=maximum,'source_file_size_bound')
    with path.open('rb') as stream:
        opened=os.fstat(stream.fileno());raw=stream.read(maximum+1);closed=os.fstat(stream.fileno())
    after=path.stat();completed=clock()
    require(started<=completed,'source_read_clock_regression')
    require(len(raw)<=maximum and len(raw)==before.st_size,'source_read_size_mismatch')
    require(identity(before)==identity(opened)==identity(closed)==identity(after),'source_changed_during_read')
    return raw,dict(path=str(path),sha256=sha(raw),bytes=len(raw),read_started_epoch=started,
        read_completed_epoch=completed,before_stat=identity(before),after_stat=identity(after),stable=True)


def save_new(path, raw):
    with path.open('xb') as stream:
        stream.write(raw);stream.flush();os.fsync(stream.fileno())
    return dict(path=str(path),sha256=sha(raw),bytes=len(raw))


def save_json(path, value):
    return save_new(path,(json.dumps(value,sort_keys=True,indent=2,allow_nan=False)+'\n').encode())


def compact_reason(exc):
    text=str(exc)
    return text if re.fullmatch('[A-Za-z_][A-Za-z0-9_]{0,100}',text) else type(exc).__name__+'_bounded_diagnostic'


def age_summary(rows, cutoff):
    values=[cutoff-r['original_known_epoch'] for r in rows]
    return dict(count=len(values),minimum_sec=min(values) if values else None,
        median_sec=statistics.median(values) if values else None,maximum_sec=max(values) if values else None)


def source_counts(core):
    # The accepted producer returns three top-level counts, not a counts wrapper.
    names=('topic_count','directional_topic_count','context_topic_count')
    require(all(type(core.get(name)) is int and core[name]>=0 for name in names),'producer_count_schema')
    return {name:core[name] for name in names}
