def clocks(minute, steps=(0,1,2)):
    times = iter(fixture.stamp(minute) + step for step in steps)
    return lambda: (next(times), fixture.clock_state(minute))


def publish(store, snapshot, minute=2, hook=None):

@pytest.mark.parametrize('mutation',['trailing','concatenated','truncated','oversized','digest'])
def test_frame_bombs_and_corruption_rejected(mutation):
    original=b'{"payload_json":"{  \\"literal\\": 1 }"}'
    size,key,packed=compact.frame(original)
    if mutation=='trailing':packed+=b'x'
    if mutation=='concatenated':packed+=zlib.compress(b'x')
    if mutation=='truncated':packed=packed[:-1]
    if mutation=='oversized':size-=1
    if mutation=='digest':key='0'*64
    with pytest.raises((ValueError,zlib.error)):compact.expand((size,key,packed))

def test_slow_cold_validation_releases_actual_reader_lock_before_writer(case,monkeypatch):
    """A writer can commit while a cold reader validates its older owned view."""
    snapshot,ctx=ready(case);source,store,path,con=case
    fixture.fixture.put(con,fixture.article(4,summary='next'),4);con.commit()
    newer=fixture.read(source,4.1,snapshot['next_checkpoint'])
    entered=threading.Event();release=threading.Event();failures=[]
    original=compact.StreamValidator.add
    def paused(self,envelope):
        if threading.current_thread().name=='owned_slow_reader':
            entered.set();assert release.wait(10)
        return original(self,envelope)
    monkeypatch.setattr(compact.StreamValidator,'add',paused)
    pub=publisher;pub._CAPTURE_CACHE.clear()
    def cold():
        try:read(store,snapshot)
        except BaseException as error:failures.append(error)
    thread=threading.Thread(target=cold,name='owned_slow_reader');thread.start()
    try:
        assert entered.wait(5)
        result=publish(store,newer,4.2)
        assert result['batch_sequence']==2
    finally:release.set();thread.join(10)
    assert not thread.is_alive() and not failures
    with sqlite3.connect(store) as db:
        assert db.execute('SELECT COUNT(*) FROM attempts').fetchone()[0]==2
        assert db.execute('SELECT COUNT(*) FROM published').fetchone()[0]==2
    # The slow earlier snapshot may replace an in-process cache entry, but the
    # next actual capture must validate and expose the current complete head.
    assert read(store,snapshot)['head']['sequence']==2
    metrics=publisher.capture_metrics(store)
    assert metrics['real_transaction_closed_before_semantic_validation'] is True
    assert metrics['transaction_seconds']<=publisher.MAX_CAPTURE_SECONDS

def test_expired_first_capture_closes_before_fresh_retry(case,monkeypatch):

    def slow_first(connection,profile,deadline):
        connections.append(connection)
        if len(connections)==1:
            original(connection,profile,deadline)
            time.sleep(.06)
            raise ValueError('packed_capture_time_bound')
        with pytest.raises(sqlite3.ProgrammingError,match='closed'):
            connections[0].execute('SELECT 1')
        return original(connection,profile,deadline)
    def validate(*args):validated.append(1);return original_validate(*args)

def test_cold_validation_allowance_does_not_expand_fresh_read_budget(case,monkeypatch):
    snapshot,ctx=ready(case);source,store,path,con=case
    original=publisher._read_owned_capture;allowances=[]
    def measured(tables,profile,deadline,base):
        allowances.append((base is None,deadline-time.monotonic()))
        return original(tables,profile,deadline,base)
    monkeypatch.setattr(publisher,'_read_owned_capture',measured)
    publisher._CAPTURE_CACHE.clear()
    cold=read(store,snapshot);warm=read(store,snapshot)

    assert not allowances[1][0] and 20<allowances[1][1]<=30
    assert publisher.capture_metrics(store)['transaction_seconds']<=publisher.MAX_CAPTURE_SECONDS
