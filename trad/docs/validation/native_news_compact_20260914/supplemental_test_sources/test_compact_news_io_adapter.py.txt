@pytest.mark.parametrize('module',[io,base])
def test_compact_archive_roundtrip_preserves_exact_lexical_strings(tmp_path,module):
    root=tmp_path/'archive';root.mkdir();archive=module.Archive(root)
    original={'schema_version':'test_only','objects':[
        [str(i),'{ "amount" : 1.000, "headline": "\\u65e5\\u672c" }',str(i)*9000]
        for i in range(110)],'metadata':{'retained_clock':12.25,'available':False}}
    key=module._store_bundle(archive,original)
    assert module._load_bundle(archive,key)==original
    count=len(archive.files);assert module._store_bundle(archive,original)==key
    assert len(archive.files)==count
    original['objects'].append(['later','same exact string','new'])
    new=module._store_bundle(archive,original)
    assert module._load_bundle(archive,new)==original and new!=key
    assert module._load_bundle(archive,key)['objects']==original['objects'][:-1]

def test_duplicate_mapping_keys_and_legacy_recipe_are_rejected(tmp_path):
    root=tmp_path/'archive';root.mkdir();archive=io.Archive(root)
    leaf=archive.put({'kind':'value','value':1})
    rows=archive.put({'kind':'value','value':[['a',leaf],['a',leaf]]})

    rootkey=archive.put({'schema_version':io.RECIPE,'compact_root':body,
        'digest_scope':'compact_manifest_not_expanded_legacy_readback'})
    with pytest.raises(ValueError,match='compact_recipe_mapping_order'):
        io._load_bundle(archive,rootkey)

    with pytest.raises(ValueError,match='capture_compact_recipe_required'):
        io._load_bundle(archive,old)


def test_original_receipt_clock_is_not_replay_or_current_read_time():
    io._original_receipts_by_read({'consumer_receipt_max_epoch':20.,'observation_count':2},21.)
    io._original_receipts_by_read({'consumer_receipt_max_epoch':None,'observation_count':0},21.)
    for value in ({'consumer_receipt_max_epoch':22.,'observation_count':2},
                  {'consumer_receipt_max_epoch':None,'observation_count':2}):
        with pytest.raises(ValueError,match='io_original_consumer_receipt_after_actual_read'):
            io._original_receipts_by_read(value,21.)


def test_replay_and_prior_failed_generation_cannot_be_current():
    session=io._make_session({'failure_serial':2,'usable':True})
    replay=io._make_capture((None,b'{}',b'{}',None,None,'immutable_replay'))
    stale=io._make_capture((None,b'{}',b'{}',session,1,'actual_capture'))
    valid=io._make_capture((None,b'{}',b'{}',session,2,'actual_capture'))
    for capture in (replay,stale):
        with pytest.raises(ValueError,match='io_capture_invalidated_or_replay_only'):
            io._eligible(session,capture)
    assert io._eligible(session,valid)[0]['usable'] is True
    io._session(session)['usable']=False
    with pytest.raises(ValueError,match='io_capture_invalidated_or_replay_only'):
        io._eligible(session,valid)

def test_hash_prefix_catalog_append_changes_only_local_nodes(tmp_path):
    root=tmp_path/'archive';root.mkdir();archive=io.Archive(root)
    def row(i):
        key=hashlib.sha256(str(i).encode()).hexdigest()
        return ['publication',key,100,key,50,key,0]
    rows=sorted((row(i) for i in range(1000)),key=lambda value:value[1])
    key=io._store_object_catalog(archive,rows);before=len(archive.files)
    assert io._load_object_catalog(archive,key,'publication')==rows
    later=sorted([*rows,*(row(i) for i in range(1000,1005))],key=lambda value:value[1])
    new=io._store_object_catalog(archive,later)
    assert len(archive.files)-before<40
    assert io._load_object_catalog(archive,new,'publication')==later
    assert io._load_object_catalog(archive,key,'publication')==rows

def test_bootstrap_is_separate_preparation_and_returns_no_current_handle(case,tmp_path,monkeypatch):
    bind_fixture_generation(monkeypatch);snapshot,_=fixtures.ready(case)
    _,store,observations,_=case;root=tmp_path/'archive';root.mkdir()
    config={'publication_path':str(store),'observation_path':str(observations),
            'archive_root':str(root),'cohort_id':'compact_new_only','consumer_id':'compact_test',
            'policy':snapshot['policy'],'input_identity':snapshot['input_identity']}
    graph={'synthetic_fixture_binding':True}
    session=io._make_session({'config':io.encode(config),'graph':graph,'failure_serial':0,'usable':True,
                             'last_failure':None,'lock':threading.RLock()})
    monkeypatch.setattr(io,'source_graph',lambda:dict(graph))
    times=iter([fixtures.fixture.stamp(4),fixtures.fixture.stamp(4)+1])
    result=io.bootstrap_inputs(session,clock=lambda:next(times))
    assert result['status']=='cache_prepared_fresh_capture_required'
    assert result['fresh_health_proven'] is False and result['capture_handle_returned'] is False
    assert result['original_availability_unchanged'] is True and io._session(session)['usable'] is False
    assert io._session(session)['failure_serial']==1
    assert io.MAX_SECONDS==30 and io.MAX_BOOTSTRAP_SECONDS==330
    times=iter([fixtures.fixture.stamp(4),fixtures.fixture.stamp(4)+331])
    with pytest.raises(ValueError,match='io_bootstrap_duration_bound'):
        io.bootstrap_inputs(session,clock=lambda:next(times))
    assert io._session(session)['usable'] is False
    assert io._session(session)['last_failure']['operation']=='bootstrap_inputs'
