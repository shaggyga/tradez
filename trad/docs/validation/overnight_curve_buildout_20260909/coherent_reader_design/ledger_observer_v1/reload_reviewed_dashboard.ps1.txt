function Hash-File([string]$Path){(Get-FileHash -LiteralPath $Path -Algorithm SHA256).Hash.ToLowerInvariant()}
function Verify-Sources {
    if((Hash-File $dashboardPath) -ne $ExpectedDashboardSha256){throw 'Untested dashboard source'}
    if((Hash-File $configPath) -ne $configSha){throw 'API specification differs from reviewed bytes'}
    $api=Get-Content -LiteralPath $configPath -Raw | ConvertFrom-Json
    foreach($binding in $api.source_bindings.PSObject.Properties){
        if((Hash-File (Join-Path $projectRoot $binding.Name)) -ne $binding.Value){throw 'API source binding mismatch'}
    }
    foreach($name in $expectedRegistries.Keys){
        $path=Join-Path (Join-Path $projectRoot 'config') $name
        if((Hash-File $path) -ne $expectedRegistries[$name]){throw 'Study registry mismatch'}
        $registry=Get-Content -LiteralPath $path -Raw | ConvertFrom-Json
        if($registry.research_only -ne $true -or $registry.can_place_orders -ne $false -or $registry.can_promote -ne $false -or $registry.can_authorize -ne $false){throw 'Study authority mismatch'}
        foreach($binding in $registry.source_bindings.PSObject.Properties){
            if($binding.Name -notmatch '^(?:[a-z0-9_]+/)*[a-z0-9_]+\.py$'){throw 'Unexpected registered source path'}
            $source=[IO.Path]::GetFullPath((Join-Path $projectRoot $binding.Name))
            if(-not $source.StartsWith($projectRoot+'\',[StringComparison]::OrdinalIgnoreCase)){throw 'Source path escape'}
            if((Hash-File $source) -ne $binding.Value){throw 'Frozen study source changed'}
        }
    }
}
function Get-ProjectProcesses {
    @(Get-CimInstance Win32_Process | Where-Object {$_.Name -match '^(python|pythonw|pwsh|powershell)(\.exe)?$'} | ForEach-Object {
        $match=[regex]::Match([string]$_.CommandLine,'(?i)(?:"([^"]+\.(?:py|ps1))"|([^\s"]+\.(?:py|ps1)))(?:\s|$)')
        $path=if($match.Groups[1].Success){$match.Groups[1].Value}else{$match.Groups[2].Value}
        if($path.StartsWith($projectRoot+'\',[StringComparison]::OrdinalIgnoreCase)){
            [pscustomobject]@{Raw=$_;ScriptPath=$path}
        }
    })
}
function Private-FreeIdentity($Row){
    [ordered]@{pid=[int]$Row.Raw.ProcessId;parent_pid=[int]$Row.Raw.ParentProcessId;name=$Row.Raw.Name;created_utc=$Row.Raw.CreationDate.ToUniversalTime().ToString('o');script_path=$Row.ScriptPath}
}
function Assert-SameProcess($Observed){
    $current=@(Get-CimInstance Win32_Process -Filter ('ProcessId = '+[int]$Observed.Raw.ProcessId))
    if($current.Count -ne 1 -or $current[0].Name -ne $Observed.Raw.Name -or $current[0].CreationDate -ne $Observed.Raw.CreationDate -or $current[0].CommandLine -ne $Observed.Raw.CommandLine){throw 'Preserved process identity changed'}
}
