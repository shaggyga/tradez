def sha(raw):return hashlib.sha256(raw).hexdigest()


def file_record(path):
    raw=path.read_bytes()
    return dict(path=str(path),sha256=sha(raw),bytes=len(raw))


def need(ok):
    if not ok:raise ValueError('validation_binding_failed')


def blocks(path):
    tree=ast.parse(path.read_bytes())
    return {n.name:ast.dump(n,include_attributes=False) for n in tree.body if isinstance(n,(ast.FunctionDef,ast.ClassDef))}
